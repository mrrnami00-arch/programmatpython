# ===== www.programmatpython.ir =====
# محل قرارگیری: ریشه سایت (public_html) کنار index.html

DirectoryIndex index.html
Options -Indexes
ErrorDocument 404 /404.html

<IfModule mod_rewrite.c>
RewriteEngine On

# 1) اجبار HTTPS و www (یک ریدایرکت ۳۰۱ واحد)
RewriteCond %{HTTPS} off [OR]
RewriteCond %{HTTP_HOST} !^www\. [NC]
RewriteRule ^ https://www.programmatpython.ir%{REQUEST_URI} [L,R=301]

# 2) حذف index.html از آدرس (جلوگیری از محتوای تکراری)
RewriteCond %{THE_REQUEST} \s/+(.*?/)?index\.html[\s?] [NC]
RewriteRule ^ /%1 [L,R=301,NE]

# 3) افزودن اسلش پایانی برای پوشه‌ها (/matlab -> /matlab/)
RewriteCond %{REQUEST_FILENAME} -d
RewriteCond %{REQUEST_URI} !/$
RewriteRule ^(.*)$ /$1/ [L,R=301]
</IfModule>

# فشرده‌سازی
<IfModule mod_deflate.c>
AddOutputFilterByType DEFLATE text/html text/css text/plain text/xml application/xml application/json application/javascript image/svg+xml
</IfModule>

# کش مرورگر
<IfModule mod_expires.c>
ExpiresActive On
ExpiresByType image/webp "access plus 1 year"
ExpiresByType image/png "access plus 1 year"
ExpiresByType image/jpeg "access plus 1 year"
ExpiresByType image/svg+xml "access plus 1 year"
ExpiresByType font/woff2 "access plus 1 year"
ExpiresByType text/css "access plus 1 month"
ExpiresByType application/javascript "access plus 1 month"
ExpiresByType text/html "access plus 0 seconds"
</IfModule>

# هدرهای امنیتی
<IfModule mod_headers.c>
Header always set Strict-Transport-Security "max-age=31536000"
Header always set X-Content-Type-Options "nosniff"
Header always set X-Frame-Options "SAMEORIGIN"
Header always set Referrer-Policy "strict-origin-when-cross-origin"
</IfModule>

# جلوگیری از دسترسی به فایل‌های حساس
<FilesMatch "^\.">
Require all denied
</FilesMatch>
